name: Actualizar datos NQ Desk

# El robot se queda trabajando en bucle: actualiza los datos cada ~5 minutos
# durante casi 6 horas y, al terminar, se vuelve a lanzar a sí mismo.
# El horario de abajo es solo una red de seguridad por si algún bucle se corta.
on:
  schedule:
    - cron: "7 * * * *"
  workflow_dispatch:
    inputs:
      periodico:
        description: "Regenerar el periódico ahora (1 = sí)"
        required: false
        default: "0"

concurrency:
  group: actualizar-datos
  cancel-in-progress: false

jobs:
  actualizar:
    runs-on: ubuntu-latest
    timeout-minutes: 355
    permissions:
      contents: write
      actions: write
    steps:
      - name: Descargar el repositorio
        uses: actions/checkout@v4

      - name: Robot en bucle (cada ~5 minutos)
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          FINNHUB_API_KEY: ${{ secrets.FINNHUB_API_KEY }}
          GEMINI_MODEL: ${{ vars.GEMINI_MODEL }}
          FIRST_FORCE: ${{ github.event.inputs.periodico }}
        run: |
          git config user.name "nq-desk-bot"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          END=$(( $(date +%s) + 340*60 ))
          FORCE="$FIRST_FORCE"
          while true; do
            git pull -q --rebase -X theirs || true
            FORCE_PERIODICO="$FORCE" python3 scripts/update.py || echo "aviso: el robot terminó con error"
            FORCE=""
            git add data
            if ! git diff --cached --quiet; then
              git commit -q -m "Datos $(date -u +'%Y-%m-%d %H:%M') UTC"
              for i in 1 2 3; do
                git pull -q --rebase -X theirs && git push -q && break
                sleep 5
              done
            fi
            [ "$(date +%s)" -ge "$END" ] && break
            sleep 270
          done

      - name: Volver a lanzarse
        if: ${{ !cancelled() }}
        env:
          GH_TOKEN: ${{ github.token }}
        run: gh workflow run actualizar.yml --ref main

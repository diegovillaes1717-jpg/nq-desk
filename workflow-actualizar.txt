name: Actualizar datos NQ Desk

on:
  schedule:
    - cron: "*/10 * * * *"   # cada 10 minutos (GitHub puede retrasarlo unos minutos)
  push:
    branches: [main]          # publica la web también cuando cambias algún archivo
    paths-ignore: ["data/**"]
  workflow_dispatch:          # botón "Run workflow" para lanzarlo a mano
    inputs:
      periodico:
        description: "Regenerar el periódico ahora (1 = sí)"
        required: false
        default: "0"

permissions:
  contents: write
  pages: write
  id-token: write

concurrency:
  group: actualizar-datos
  cancel-in-progress: false

jobs:
  actualizar:
    runs-on: ubuntu-latest
    timeout-minutes: 12
    environment:
      name: github-pages
      url: ${{ steps.deploy.outputs.page_url }}
    steps:
      - name: Descargar el repositorio
        uses: actions/checkout@v4

      - name: Ejecutar el robot
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          FINNHUB_API_KEY: ${{ secrets.FINNHUB_API_KEY }}
          GEMINI_MODEL: ${{ vars.GEMINI_MODEL }}
          FORCE_PERIODICO: ${{ github.event.inputs.periodico }}
        run: python3 scripts/update.py

      - name: Guardar los datos nuevos en el repositorio
        run: |
          git config user.name "nq-desk-bot"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data
          if git diff --cached --quiet; then
            echo "Sin cambios"
          else
            git commit -m "Datos $(date -u +'%Y-%m-%d %H:%M') UTC"
            for i in 1 2 3; do
              git pull --rebase -X theirs && git push && break
              sleep 5
            done
          fi

      - name: Preparar la web
        run: |
          mkdir -p _site
          cp -r index.html assets data _site/
          touch _site/.nojekyll

      - name: Subir la web
        uses: actions/upload-pages-artifact@v3
        with:
          path: _site

      - name: Publicar en GitHub Pages
        id: deploy
        uses: actions/deploy-pages@v4
